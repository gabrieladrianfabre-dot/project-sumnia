\setlength{\columnseprule}{0.2pt}
\raggedcolumns
\setlength{\parindent}{0pt}

\setlength{\headheight}{14pt}

% Difficulty colours and tag
\definecolor{tagE}{HTML}{2E7D32}
\definecolor{tagA}{HTML}{1565C0}
\definecolor{tagD}{HTML}{C62828}
\newcommand{\qatag}[1]{\textsuperscript{\textbf{\textcolor{tag#1}{#1}}}}

% Card counters (reset per grade/part with \resetcards, printed with \cardstats)
\newcounter{qaE}\newcounter{qaA}\newcounter{qaD}
\newcommand{\resetcards}{\setcounter{qaE}{0}\setcounter{qaA}{0}\setcounter{qaD}{0}}
\newcommand{\cardstats}{\par\medskip\noindent{\footnotesize\itshape Cards in this part:
  \number\numexpr\value{qaE}+\value{qaA}+\value{qaD}\relax\
  (E~\arabic{qaE}, A~\arabic{qaA}, D~\arabic{qaD})}\par}

% \qa{question}{answer}{E|A|D}
\newcommand{\qa}[3]{\par\noindent\hangindent=1em\hangafter=1
  \stepcounter{qa#3}%
  \textbf{Q:}~#1~$\rightarrow$~\textbf{A:}~\textit{#2}\,\qatag{#3}\par\vspace{1pt}}

% \verify{text}: prints text with a visible [VERIFY] tag and logs it for the final list
\newcommand{\verify}[1]{#1\textsuperscript{\,\textcolor{red}{\textbf{[VERIFY]}}}%
  \addtocontents{vfy}{\protect\vfyline{\unexpanded{#1}}{\thepage}}}
\newcommand{\vfyline}[2]{\par\noindent\hangindent=1em\hangafter=1
  \textbullet~#1\dotfill p.\,#2\par}
\makeatletter
\newcommand{\listofverify}{%
  \section*{Items to verify}
  \markboth{Items to verify}{}
  \addcontentsline{toc}{section}{Items to verify}
  {\small Every \textcolor{red}{\textbf{[VERIFY]}} tag in this document, with its page.
  These are facts that were uncertain, that changed recently, or where sources disagree.
  Check each against the source named in \emph{Sources} (or ask your coach) before the contest.\par\medskip}
  {\small\@starttoc{vfy}}}
\makeatother
