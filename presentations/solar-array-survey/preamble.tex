% ---------------------------------------------------------------------------
% preamble.tex — packages, theme, and the TikZ styles shared by the figures.
% ---------------------------------------------------------------------------

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lmodern}
\usepackage{amsmath}
\usepackage{booktabs}
\usepackage{tikz}
\usetikzlibrary{calc,arrows.meta,positioning}

\usetheme{Madrid}
% Computer Modern serif. Comment this line out for Beamer's sans default.
\usefonttheme{serif}

% --- Figure palette -------------------------------------------------------
% Kept to the theme's own two colours so the diagrams sit inside the deck
% rather than next to it: structure blue for the survey data, red for the
% quantity being derived, greys for everything supporting.
\definecolor{figink}{RGB}{0,0,0}
\definecolor{figmuted}{RGB}{105,105,110}
\definecolor{figaccent}{RGB}{51,51,178}    % beamer's structure blue
\definecolor{figalert}{RGB}{170,30,30}
\definecolor{figpale}{RGB}{237,237,245}
\definecolor{figrule}{RGB}{170,170,175}

% --- Shared TikZ styles ---------------------------------------------------
\tikzset{
  beam/.style   = {figink, line width=1.1pt, line cap=round},
  ray/.style    = {figaccent, line width=0.9pt, line cap=round},
  north/.style  = {figmuted, line width=0.7pt, dashed, -{Stealth[length=4pt]}},
  helper/.style = {figrule, line width=0.6pt},
  vertex/.style = {circle, fill=figink, inner sep=1.3pt},
  vlabel/.style = {figink, font=\small\bfseries},
  dim/.style    = {figmuted, font=\scriptsize},
  angmark/.style= {figalert, line width=0.9pt},
  panelcell/.style = {fill=figaccent!75, draw=white, line width=0.3pt},
}
