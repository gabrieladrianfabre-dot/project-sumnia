% === Where we are going ===================================================
\begin{frame}{The brief}{One number decides the whole design}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      An engineering firm is scoping a small solar array in an unused
      triangular corner of a warehouse rooftop, boxed in by two existing
      structural beams. They need \amber{the enclosed area} --- that is what
      sets how many panels the site can hold.
      \medskip

      \panel{%
        The catch: the surveyor never recorded an angle. The field notes
        contain \teal{two bearings}, which is how rooftop and land surveys
        are actually reported. So the job splits in two --- recover the
        angle, then find the area.}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{enumerate}
        \item The site and the field notes
        \item Bearings $\rightarrow$ included angle
        \item The triangle, reduced to SAS
        \item Why $\tfrac12ab\sin C$ works
        \item The calculation
        \item The answer, and what it is worth
      \end{enumerate}
    \end{column}
  \end{columns}
  \note{Open on the practical question, not the formula. The audience should
  want the area before we hand them a method for finding it.}
\end{frame}

\begin{frame}{The site}{Plan view of the rooftop corner}

% === The field notes ======================================================
\begin{frame}{What the surveyor actually recorded}{Total station, set up over point O}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \panel{%
        \small
        \begin{tabular}{@{}ll@{}}
          \multicolumn{2}{@{}l}{\bfseries Field notes --- station O}\\[3pt]
          \midrule
          Beam OA & 34.6 m at bearing $038^\circ$\\
          Beam OB & 27.2 m at bearing $142^\circ$\\
          \midrule
          Angle at O & \emph{not recorded}\\
        \end{tabular}}
      \medskip

      \small A total station measures \teal{distance and direction}, and
      direction is logged as a bearing --- nobody stands in the corner with a
      protractor. \amber{So the included angle has to be derived.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \vspace{-0.2cm}
      \scalebox{0.7}{\diagBearingConvention}
    \end{column}
  \end{columns}
  \note{Budget a beat here. The audience needs to accept that "no angle given"
  is normal survey practice, not a trick in the question.}
\end{frame}

% === Deriving the angle ===================================================
\begin{frame}{From two bearings to one angle}{Both are measured from the same point, in the same sense}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \small
      Two conditions make this a plain subtraction:
      \begin{itemize}
        \item both bearings share the \teal{same reference point}, O;
        \item both are measured \teal{clockwise from north}.
      \end{itemize}
      \medskip

      \keypanel{\centering $\angle AOB = 142^\circ - 38^\circ = \mathbf{104^\circ}$}
      \medskip

      No trigonometry yet --- this step is bookkeeping about how the
      directions were recorded.
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \vspace{-0.3cm}
      \scalebox{0.78}{\diagBearingSubtraction}
    \end{column}
  \end{columns}
  \note{This is the slide that makes the scenario real rather than
  plug-and-chug. Do not rush it.}
\end{frame}

% === The general rule =====================================================
\begin{frame}{The rule in general}{Subtraction is not always the whole story}
  \centering
  \scalebox{0.86}{\diagBearingRule}

  \vspace{0.25cm}
  {\small Subtract the bearings; if the result exceeds $180^\circ$ you have
  measured the \emph{reflex} side, so take $360^\circ$ minus it instead.
  Here $104^\circ < 180^\circ$, so no adjustment is needed.}
  \note{Include this even though our case does not need it --- it shows the
  method was applied knowingly rather than by luck.}
\end{frame}

% === SAS ==================================================================
\begin{frame}{Now it is a standard SAS triangle}{Two sides and the angle between them}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \vspace{0.2cm}
      \panel{%
        \begin{tabular}{@{}ll@{}}
          $a$ & $=$ OB $= 27.2$ m\\[3pt]
          $b$ & $=$ OA $= 34.6$ m\\[3pt]
          $C$ & $= 104^\circ$ (derived)\\
        \end{tabular}}
      \medskip

      \small The angle sits \amber{between} the two known sides --- exactly the
      SAS pattern.
      \medskip

      $C$ is obtuse, so the triangle is genuinely oblique: no right angle to
      fall back on, and no perpendicular height in the field notes.
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \vspace{-0.15cm}
      \scalebox{0.8}{\diagSAS}
    \end{column}
  \end{columns}
\end{frame}

% === Why the formula works ================================================
\begin{frame}{Why $\tfrac12ab\sin C$ is the right tool}{$\sin C$ recovers the height nobody measured}
  \centering
  \vspace{-0.1cm}
  \scalebox{0.86}{\diagHeight}

  \vspace{-0.15cm}
  {\small The familiar $\tfrac12\times\text{base}\times\text{height}$ never went
  away. \amber{$\sin C$ simply supplies the height} from the side and angle we
  do have.}
  \note{The one line worth saying out loud: sin C substitutes for a
  perpendicular height that was never directly measured.}
\end{frame}

% === The calculation ======================================================
\begin{frame}{The calculation}{Substitute once, then evaluate}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \vspace{0.1cm}
      \begin{align*}
        \text{Area} &= \tfrac12\,ab\sin C\\[4pt]
                    &= \tfrac12 \times 34.6 \times 27.2 \times \sin 104^\circ\\[4pt]
                    &= 470.56 \times \sin 104^\circ\\[4pt]
                    &= 470.56 \times 0.9703\\[4pt]
                    &\approx \amber{456.58\ \text{m}^2}
      \end{align*}
    \end{column}
    \begin{column}{0.4\textwidth}
      \vspace{0.3cm}
      \panel{%
        \small
        \textbf{Watch for}\\[4pt]
        Calculator in \teal{degrees}, not radians.\\[6pt]
        $\sin 104^\circ = \sin 76^\circ = 0.9703$ --- an obtuse angle gives a
        perfectly ordinary positive sine.\\[6pt]
        Round only at the \teal{final step}.}
    \end{column}
  \end{columns}
\end{frame}

% === Accuracy note ========================================================
\begin{frame}{A note on rounding}{Where the last decimal comes from}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \vspace{0.2cm}
      $\sin 104^\circ = 0.970296\ldots$, so
      \[470.56 \times 0.970296 = 456.582\ldots\]
      \medskip

      A rounded $0.9709$ would give $456.87\ \text{m}^2$ --- about
      $0.3\ \text{m}^2$ adrift, roughly \amber{one panel's worth} of area.
      \medskip

      Small, but the kind of error that compounds: carry the full value
      through and round once, at the end.
    \end{column}
    \begin{column}{0.45\textwidth}
      \vspace{0.4cm}
      \panel{%
        \small
        \begin{tabular}{@{}ll@{}}
          \multicolumn{2}{@{}l}{\bfseries Sensitivity of the answer}\\[4pt]
          \midrule
          $\sin C = 0.9703$ & $456.58\ \text{m}^2$\\
          $\sin C = 0.9709$ & $456.87\ \text{m}^2$\\
          \midrule
          \multicolumn{2}{@{}l}{\scriptsize difference: $0.29\ \text{m}^2$}\\
        \end{tabular}}
    \end{column}
  \end{columns}
  \note{Optional slide --- drop it if time is short, but it is cheap
  credibility if a marker checks the arithmetic.}
\end{frame}

% === Takeaways ============================================================
\begin{frame}{What to take away}{Two ideas, one method}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \panel{%
        \textbf{1. The angle was earned, not given}\\[5pt]
        Bearings share a reference direction and a rotational sense, so their
        difference \emph{is} the angle between the lines. Real survey data
        arrives this way; converting it is part of the problem.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \keypanel{%
        \textbf{2. It is still just base $\times$ height}\\[5pt]
        $\tfrac12ab\sin C$ is not a new formula to memorise. $\sin C$ stands in
        for a perpendicular height that was never measured --- which is why
        SAS is enough.}
    \end{column}
  \end{columns}
  \vspace{0.5cm}
  \centering
  {\large From \teal{two distances and two directions} to
   \amber{$456.58\ \text{m}^2$} --- and a panel count.}
\end{frame}
